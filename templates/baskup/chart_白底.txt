        type: 'pie',

        title: {

      // 柱状图数据

        x: filteredLabels, // 使用过滤后的标签

        type: 'bar',
        marker: {
          color: '#ffa600' /* 橙色柱状图 */

      // 生成Top 5列表
      function generateTop5(labels, values) {
        var data = labels.map((label, index) => {
          return { label: label, value: values[index] };
        });

        data.sort((a, b) => b.value - a.value);

        var top5 = data.slice(0, 5);

        var top5List = document.getElementById('top5-list');
        top5List.innerHTML = '';

        top5.forEach(item => {
          var listItem = document.createElement('li');
          listItem.textContent = `${item.label}: ${item.value}`;
          top5List.appendChild(listItem);
        });
      }

      generateTop5(labels, values);
