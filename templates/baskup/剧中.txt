        function fetchDataAndRenderTable() {
          $.ajax({
            url: '/projects',
            method: 'GET',
            success: function(data) {
              if (Array.isArray(data) && data.length > 0) {
                renderTable(data);
              } else {
                console.error('Data format is incorrect or empty');
                layer.msg('数据库中尚未添加数据', { icon: 2, time: 2000 });
                renderTable([]);
              }
            },
            error: function() {
              layer.msg('Failed to fetch data from server', { icon: 2, time: 2000 });
              renderTable([]);
            }
          });
        }
        
        function renderTable(data) {
          table.render({
            elem: '#projectboard',
            data: data,
            page: true, // 启用分页
            limit: 10, // 每页显示的条数
            limits: [10, 20, 30, 50], // 可供选择的每页条数选项
            cols: [[
